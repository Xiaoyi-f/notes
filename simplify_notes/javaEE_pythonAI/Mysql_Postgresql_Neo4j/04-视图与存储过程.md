# SQL 学习笔记 - 视图与存储过程

## 一、视图（VIEW）

### 1.1 什么是视图

视图是虚拟表，不实际存储数据，只是保存了查询语句。每次查询视图时，数据库都会执行保存的查询，从基表中获取最新数据。

**核心价值：**
1. 简化复杂查询
2. 数据安全与权限控制
3. 保持逻辑一致性

---

### 1.2 创建视图

```sql
-- 基本语法
CREATE VIEW 视图名 AS
SELECT 语句;

-- 示例：员工部门视图
CREATE VIEW employee_department_view AS
SELECT 
    e.id,
    e.name,
    e.email,
    d.department_name,
    e.salary
FROM employees e
JOIN departments d ON e.department_id = d.id;

-- 使用视图（和普通表一样）
SELECT * FROM employee_department_view;
SELECT * FROM employee_department_view WHERE department_name = 'IT';

-- 带条件的视图
CREATE VIEW it_employees_view AS
SELECT 
    e.id,
    e.name,
    e.email,
    d.department_name,
    e.salary
FROM employees e
JOIN departments d ON e.department_id = d.id
WHERE d.department_name = 'IT';
```

### 1.3 视图应用场景

#### 场景1：简化复杂查询

```sql
-- 原始复杂查询
SELECT 
    u.name,
    u.email,
    o.order_id,
    o.order_date,
    oi.product_id,
    p.product_name,
    oi.quantity,
    oi.price
FROM users u
JOIN orders o ON u.id = o.user_id
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.id;

-- 创建视图简化
CREATE VIEW order_detail_view AS
SELECT 
    u.name,
    u.email,
    o.order_id,
    o.order_date,
    oi.product_id,
    p.product_name,
    oi.quantity,
    oi.price
FROM users u
JOIN orders o ON u.id = o.user_id
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.id;

-- 简单使用
SELECT * FROM order_detail_view WHERE name = '张三';
```

---

#### 场景2：数据安全与权限控制

```sql
-- 隐藏敏感信息
CREATE VIEW employee_public_view AS
SELECT 
    id,
    name,
    department_id,
    position
FROM employees;

-- 授予访问视图的权限，而不是原始表
GRANT SELECT ON employee_public_view TO some_user;

-- 用户只能看到视图中的字段，无法看到 salary、phone 等敏感信息
```

---

#### 场景3：聚合数据视图

```sql
-- 创建部门统计视图
CREATE VIEW department_stats_view AS
SELECT 
    d.id AS department_id,
    d.department_name,
    COUNT(e.id) AS employee_count,
    AVG(e.salary) AS avg_salary,
    SUM(e.salary) AS total_salary,
    MIN(e.salary) AS min_salary,
    MAX(e.salary) AS max_salary
FROM departments d
LEFT JOIN employees e ON d.id = e.department_id
GROUP BY d.id, d.department_name;

-- 使用
SELECT * FROM department_stats_view WHERE employee_count > 10;
```

### 1.6 视图的限制

**性能问题：**
- 每次查询都执行底层的SQL
- 复杂视图可能很慢

**更新限制：**
- 简单视图（单表、无聚合）可以更新
- 复杂视图（多表连接、聚合函数、DISTINCT、GROUP BY）通常只读

```sql
-- 简单视图可以更新
CREATE VIEW simple_view AS
SELECT id, name, email FROM users;

-- 可以更新
UPDATE simple_view SET email = 'new@example.com' WHERE id = 1;

-- 复杂视图通常只读
CREATE VIEW complex_view AS
SELECT department, COUNT(*) AS emp_count
FROM employees
GROUP BY department;

-- 可能无法更新
UPDATE complex_view SET emp_count = 10;  -- ❌ 错误
```

---

### 1.7 视图 vs 临时表

| 特性 | 视图（View） | 临时表（Temporary Table） |
|------|-------------|-------------------------|
| **本质** | 虚拟表，不存数据 | 真实表，存数据 |
| **生命周期** | 手动删除 | 会话结束自动删除 |
| **数据** | 始终是最新的 | 创建后是静态的 |
| **主要用途** | 简化查询、安全控制 | 中间结果处理、性能优化 |

```sql
-- 临时表（MySQL）
CREATE TEMPORARY TABLE temp_orders AS
SELECT * FROM orders WHERE order_date >= '2024-01-01';
-- 会话结束后自动删除

-- 临时表（PostgreSQL）
CREATE TEMP TABLE temp_orders AS
SELECT * FROM orders WHERE order_date >= '2024-01-01';
```

## 二、存储过程（Stored Procedure）

### 2.1 什么是存储过程

存储过程是预编译并存储在数据库中的SQL代码块，可以包含复杂业务逻辑、流程控制和多次执行的SQL语句。

**核心优势：**
1. 性能提升：预编译缓存
2. 减少网络流量：一次调用执行多句SQL
3. 安全性：权限控制
4. 代码复用
5. 事务控制

---

### 2.2 MySQL 存储过程语法

#### 基本语法

```sql
DELIMITER $$

CREATE PROCEDURE 存储过程名([参数列表])
BEGIN
    -- SQL 语句
END$$

DELIMITER ;

-- 调用
CALL 存储过程名([参数]);

-- 删除
DROP PROCEDURE IF EXISTS 存储过程名;
```

#### 无参存储过程

```sql
DELIMITER $$
CREATE PROCEDURE GetActiveUsers()
BEGIN
    SELECT * FROM users WHERE status = 'active';
END$$
DELIMITER ;

-- 调用
CALL GetActiveUsers();
```

#### 带输入参数

```sql
DELIMITER $$
CREATE PROCEDURE GetUserById(IN userId INT)
BEGIN
    SELECT * FROM users WHERE id = userId;
END$$
DELIMITER ;

-- 调用
CALL GetUserById(1);
```

#### 带输出参数

```sql
DELIMITER $$
CREATE PROCEDURE GetUserCount(IN deptId INT, OUT totalCount INT)
BEGIN
    SELECT COUNT(*) INTO totalCount FROM users WHERE department_id = deptId;
END$$
DELIMITER ;

-- 调用
CALL GetUserCount(10, @count);
SELECT @count AS total_users;
```

#### 带输入输出参数

```sql
DELIMITER $$
CREATE PROCEDURE IncrementCounter(INOUT counter INT)
BEGIN
    SET counter = counter + 1;
END$$
DELIMITER ;

-- 调用
SET @counter = 5;
CALL IncrementCounter(@counter);
SELECT @counter;  -- 6
```

### 2.4 流程控制

#### IF 条件语句

```sql
-- MySQL
DELIMITER $$
CREATE PROCEDURE UpdateSalary(IN empId INT, IN rate DECIMAL)
BEGIN
    DECLARE years INT;
    
    SELECT years_of_service INTO years FROM employees WHERE id = empId;
    
    IF years > 5 THEN
        UPDATE employees SET salary = salary * (1 + rate * 1.5) WHERE id = empId;
    ELSEIF years > 2 THEN
        UPDATE employees SET salary = salary * (1 + rate) WHERE id = empId;
    ELSE
        UPDATE employees SET salary = salary * (1 + rate * 0.5) WHERE id = empId;
    END IF;
END$$
DELIMITER ;

-- PostgreSQL
CREATE OR REPLACE FUNCTION update_salary(emp_id INT, rate NUMERIC)
RETURNS VOID AS $$
DECLARE
    years INT;
BEGIN
    SELECT years_of_service INTO years FROM employees WHERE id = emp_id;
    
    IF years > 5 THEN
        UPDATE employees SET salary = salary * (1 + rate * 1.5) WHERE id = emp_id;
    ELSIF years > 2 THEN
        UPDATE employees SET salary = salary * (1 + rate) WHERE id = emp_id;
    ELSE
        UPDATE employees SET salary = salary * (1 + rate * 0.5) WHERE id = emp_id;
    END IF;
END;
$$ LANGUAGE plpgsql;
```

#### WHILE 循环

```sql
-- MySQL
DELIMITER $$
CREATE PROCEDURE GenerateNumbers(IN n INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    CREATE TEMPORARY TABLE IF NOT EXISTS numbers (num INT);
    
    WHILE i <= n DO
        INSERT INTO numbers VALUES (i);
        SET i = i + 1;
    END WHILE;
    
    SELECT * FROM numbers;
END$$
DELIMITER ;

-- PostgreSQL
CREATE OR REPLACE FUNCTION generate_numbers(n INT)
RETURNS SETOF INT AS $$
DECLARE
    i INT := 1;
BEGIN
    WHILE i <= n LOOP
        RETURN NEXT i;
        i := i + 1;
    END LOOP;
    RETURN;
END;
$$ LANGUAGE plpgsql;
```

---

#### REPEAT 循环（MySQL）

```sql
DELIMITER $$
CREATE PROCEDURE SumUntil(IN target INT)
BEGIN
    DECLARE sum_val INT DEFAULT 0;
    DECLARE i INT DEFAULT 1;
    
    REPEAT
        SET sum_val = sum_val + i;
        SET i = i + 1;
    UNTIL sum_val > target END REPEAT;
    
    SELECT sum_val, i;
END$$
DELIMITER ;
```

---

#### FOR 循环（PostgreSQL）

```sql
CREATE OR REPLACE FUNCTION sum_until(target INT)
RETURNS INT AS $$
DECLARE
    sum_val INT := 0;
BEGIN
    FOR i IN 1..100 LOOP
        sum_val := sum_val + i;
        EXIT WHEN sum_val > target;
    END LOOP;
    RETURN sum_val;
END;
$$ LANGUAGE plpgsql;
```

### 2.5 事务处理

```sql
-- MySQL 转账存储过程
DELIMITER $$
CREATE PROCEDURE TransferMoney(
    IN fromAccount INT,
    IN toAccount INT,
    IN amount DECIMAL(10,2)
)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SELECT '转账失败' AS result;
    END;
    
    START TRANSACTION;
    
    -- 扣款
    UPDATE accounts SET balance = balance - amount WHERE id = fromAccount;
    
    -- 检查余额
    IF (SELECT balance FROM accounts WHERE id = fromAccount) < 0 THEN
        ROLLBACK;
        SELECT '余额不足' AS result;
    ELSE
        -- 收款
        UPDATE accounts SET balance = balance + amount WHERE id = toAccount;
        COMMIT;
        SELECT '转账成功' AS result;
    END IF;
END$$
DELIMITER ;

-- PostgreSQL 版本
CREATE OR REPLACE FUNCTION transfer_money(
    from_account INT,
    to_account INT,
    amount NUMERIC(10,2)
) RETURNS VARCHAR AS $$
BEGIN
    -- 扣款
    UPDATE accounts SET balance = balance - amount WHERE id = from_account;
    
    -- 检查余额
    IF (SELECT balance FROM accounts WHERE id = from_account) < 0 THEN
        RAISE EXCEPTION '余额不足';
    END IF;
    
    -- 收款
    UPDATE accounts SET balance = balance + amount WHERE id = to_account;
    
    RETURN '转账成功';
EXCEPTION
    WHEN OTHERS THEN
        RAISE NOTICE '转账失败: %', SQLERRM;
        RETURN '转账失败';
END;
$$ LANGUAGE plpgsql;
```

### 2.6 游标（Cursor）

```sql
-- MySQL 游标
DELIMITER $$
CREATE PROCEDURE ProcessEmployees()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE emp_id INT;
    DECLARE emp_name VARCHAR(50);
    DECLARE emp_salary DECIMAL(10,2);
    
    -- 声明游标
    DECLARE cur CURSOR FOR 
        SELECT id, name, salary FROM employees WHERE department_id = 1;
    
    -- 声明结束处理程序
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    -- 创建临时表存储结果
    CREATE TEMPORARY TABLE IF NOT EXISTS processed_emp (
        id INT,
        name VARCHAR(50),
        old_salary DECIMAL(10,2),
        new_salary DECIMAL(10,2)
    );
    
    OPEN cur;
    
    read_loop: LOOP
        FETCH cur INTO emp_id, emp_name, emp_salary;
        
        IF done THEN
            LEAVE read_loop;
        END IF;
        
        -- 处理逻辑
        DECLARE new_salary DECIMAL(10,2);
        SET new_salary = emp_salary * 1.1;
        
        -- 更新数据
        UPDATE employees SET salary = new_salary WHERE id = emp_id;
        
        -- 记录结果
        INSERT INTO processed_emp VALUES (emp_id, emp_name, emp_salary, new_salary);
    END LOOP;
    
    CLOSE cur;
    
    -- 返回结果
    SELECT * FROM processed_emp;
END$$
DELIMITER ;

-- PostgreSQL 游标
CREATE OR REPLACE FUNCTION process_employees()
RETURNS TABLE(id INT, name VARCHAR, old_salary NUMERIC, new_salary NUMERIC) AS $$
DECLARE
    emp_record RECORD;
    new_salary NUMERIC;
BEGIN
    FOR emp_record IN SELECT id, name, salary FROM employees WHERE department_id = 1
    LOOP
        -- 处理逻辑
        new_salary := emp_record.salary * 1.1;
        
        -- 更新数据
        UPDATE employees SET salary = new_salary WHERE id = emp_record.id;
        
        -- 返回结果
        RETURN QUERY SELECT emp_record.id, emp_record.name, 
                                emp_record.salary, new_salary;
    END LOOP;
    RETURN;
END;
$$ LANGUAGE plpgsql;
```
