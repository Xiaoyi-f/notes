## 三、Pinia 状态管理

```javascript
// stores/user.js
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useUserStore = defineStore('user', () => {
  // State
  const token = ref(uni.getStorageSync('token') || '')
  const userInfo = ref(uni.getStorageSync('userInfo') || null)
  
  // Getters
  const isLogin = computed(() => !!token.value)
  const userName = computed(() => userInfo.value?.name || '游客')
  
  // Actions
  const setToken = (val) => {
    token.value = val
    uni.setStorageSync('token', val)
  }
  
  const setUserInfo = (val) => {
    userInfo.value = val
    uni.setStorageSync('userInfo', val)
  }
  
  const logout = () => {
    token.value = ''
    userInfo.value = null
    uni.removeStorageSync('token')
    uni.removeStorageSync('userInfo')
  }
  
  return {
    token,
    userInfo,
    isLogin,
    userName,
    setToken,
    setUserInfo,
    logout
  }
})
```


## 四、常用工具函数

```javascript
// utils/validate.js
export const validators = {
  // 手机号
  phone: (val) => /^1[3-9]\d{9}$/.test(val),
  
  // 邮箱
  email: (val) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(val),
  
  // 身份证
  idCard: (val) => /^\d{15}$|^\d{18}$|^\d{17}[\dXx]$/.test(val),
  
  // 密码（6-20位，至少包含字母和数字）
  password: (val) => /^(?=.*[A-Za-z])(?=.*\d)[A-Za-z\d]{6,20}$/.test(val),
  
  // 金额
  money: (val) => /^\d+(\.\d{1,2})?$/.test(val),
  
  // 非空
  required: (val) => {
    if (Array.isArray(val)) return val.length > 0
    if (typeof val === 'object' && val !== null) return Object.keys(val).length > 0
    return !!val && String(val).trim() !== ''
  }
}

// utils/common.js
export const formatDate = (date, format = 'YYYY-MM-DD HH:mm:ss') => {
  const d = new Date(date)
  const year = d.getFullYear()
  const month = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  const hour = String(d.getHours()).padStart(2, '0')
  const minute = String(d.getMinutes()).padStart(2, '0')
  const second = String(d.getSeconds()).padStart(2, '0')
  
  return format
    .replace('YYYY', year)
    .replace('MM', month)
    .replace('DD', day)
    .replace('HH', hour)
    .replace('mm', minute)
    .replace('ss', second)
}

export const debounce = (fn, delay = 300) => {
  let timer = null
  return (...args) => {
    if (timer) clearTimeout(timer)
    timer = setTimeout(() => fn(...args), delay)
  }
}

export const throttle = (fn, interval = 300) => {
  let lastTime = 0
  return (...args) => {
    const now = Date.now()
    if (now - lastTime >= interval) {
      lastTime = now
      fn(...args)
    }
  }
}
```

## 五、开发规范

### 命名规范

| 类型 | 规范 | 示例 |
|------|------|------|
| 组件 | 大驼峰 | `AppButton.vue`, `UserCard.vue` |
| 页面 | 小写+连字符 | `goods-detail.vue`, `order-list.vue` |
| API函数 | 小驼峰 | `getUserInfo`, `createOrder` |
| 常量 | 大写+下划线 | `API_BASE_URL`, `MAX_PAGE_SIZE` |
| 变量 | 小驼峰 | `userList`, `isLoading` |
| 类名 | 大驼峰 | `RequestError`, `ApiClient` |

```vue
  loading.value = true

  } finally {
    loading.value = false
  }
}

const addToCart = async () => {
  if (!userStore.isLogin) {
    uni.navigateTo({ url: '/pages/login/login' })
    return
  }
  // ...
}

// ========== 暴露给模板 ==========
// 无需暴露，script setup自动暴露
</script>

<style scoped lang="scss">
.page-name {
  // 页面根样式
}
</style>
```
