## 数据库体系结构

三层模式
外模式(External): 用户级 用户能够看到和处理的数据的结构描述,局部逻辑 
模式: 概念级 数据库中全部数据的逻辑结构和特征
内模式(Internal): 物理级 存储模式,存储在存储介质上的数据的结构描述,是一种存储结构 

两层关联 
外模式关联模式
内模式关联模式

逻辑数据和物理数据的独立性有利于数据库的安全性 

单用户结构 在用户自己计算机上安装数据库,单用户使用自己的数据库 
主从式结构 数据库安装在一个主机上关联其他计算机用户 
分布式结构 C/S结构 B/S结构 数据在实际的物理分布上是分散的

## 数据模型 
概念模型: 面向用户、面向客观世界的概念化结构 
逻辑模型: 面向数据库系统的模型,具体的DBMS所支持的数据模型 开发实现数据库
物理模型: 面向计算机物理表示的模型,描述数据存储在介质上的组织结构 数据库底层实现


层次(树状)模型 网状模型 关系型数据模型 

实现ER图(实体联系图)有利于数据库的设计 --> 一对一、一对多、多对多
分析需求 -> 分析业务 -> 确立类别与相关属性信息,建立类别和相关属性信息逻辑表 -> ERD实体联系图(注意用户也是使用,使用行为/联系进行实体关联) -> DFD数据流图 -> 建表/库与检查 -> 接入使用 
Tip: 相关备注内容、细节和注意事项在做表/作图时候在表/图中适当位置标明

## 数据库规范化 
常见逻辑问题: 数据冗余、插入异常、更新异常、删除异常、属性冲突、命名冲突、结构冲突
设计概念:
自顶向下: 首先定义全局概念结构的框架,再逐步细化 
自底向上: 首先定义各局部应用的概念结构,然后将他们集成起来,得到全局概念结构 
逐步扩张: 首先定义最重要的核心概念结构,然后向外扩张,以滚雪球的方式逐步生成其他 
混合策略: 自顶向下和自底向上相结合

## Mysql 基本使用 
-- 删除数据库(不存在不报错)
DROP DATABASE IF EXISTS `xxx_db`;
-- 创建数据库
CREATE DATABASE `xxx_db`;
-- 切换使用数据库
USE `xxx_db`;

-- 查看所有数据库
SHOW DATABASES;

-- 创建表规范写法
CREATE TABLE `table_name` (
    `column_name` column_type [PRIMARY KEY | NOT NULL | DEFAULT value | UNIQUE] AUTO_INCREMENT COMMENT 'column comment',
    CONSTRAINT `constraint_name` 
    FOREIGN KEY (`column_name`) 
    REFERENCES `referenced_table_name` (`referenced_column_name`)
    [ON DELETE CASCADE] [ON UPDATE CASCADE]
);

-- 删除外键约束
ALTER TABLE `table_name` DROP FOREIGN KEY `constraint_name`;
-- 修改表名
ALTER TABLE `table_name` RENAME TO `new_table_name`;
-- 删除字段
ALTER TABLE `table_name` DROP COLUMN `column_name`;
-- 修改字段类型与约束
ALTER TABLE `table_name` MODIFY COLUMN `column_name` column_type; 
-- 新增字段，可指定位置
ALTER TABLE `table_name` ADD COLUMN `column_name` column_type [约束条件] [FIRST | AFTER `column_name`];

-- 插入多条数据
INSERT INTO `table_name` 
(`column_name1`, `column_name2`, ...) 
VALUES 
('value1', 'value2', ...), 
('value1', 'value2', ...), 
...;

-- 删除数据 
DELETE FROM `table_name` WHERE `column_name` = 'value';

-- 查询表全部字段
SELECT * FROM `table_name` LIMIT n;
-- 查询指定字段
SELECT `column_name1`, `column_name2`, ... FROM `table_name`;
-- 条件查询 条件符: = < > != <= >= BWTWEEN AND OR NOT IN(val1, val2, ...) LIKE REGEXP 
SELECT * FROM `table_name` WHERE `column_name` 条件符 'value';
-- 修改数据 
UPDATE `table_name` SET `column_name` = 'value' WHERE `column_name` = 'value';

-- SQL函数 
SELECT AVG(`column_name`) AS '平均值' FROM `table_name` WHERE condition;
SELECT COUNT(`column_name`) AS '数量' FROM `table_name` WHERE condition;
SELECT MAX(`column_name`) AS '最大值' FROM `table_name` WHERE condition;
SELECT MIN(`column_name`) AS '最小值' FROM `table_name` WHERE condition;
SELECT SUM(`column_name`) AS '和' FROM `table_name` WHERE condition;
SELECT DATE_FORMAT(`column_name`, '%Y-%m-%d %H:%i:%s');
SELECT `columna_name`, SUM(`column_name`) FROM `table_name` GROUP BY `column_name`;
SELECT ·column_name`, SUM(`column_name`) FROM `table_name` GROUP BY `column_name` HAVING 判断式; 

SELECT * FROM table_name ORDER BY column_name [ASC | DESC];

SELECT DISTINCT ...;
SELECT COUNT(DISTINCT column_name) ...;

-- 索引 有空间和查找开销 需要维护
CREATE INDEX 'idx_table_name_column_name' ON `table_name` (`column_name1`, `column_name2`, ...)
SHOW INDEX FROM `table_name` 
DROP INDEX 'index_name' ON `table_name`

-- 多表查询 
SELECT * FROM table_name1 INNER JOIN table_name2 ON table_name1.column_name = table_name2.column_name;
-- 左右边对应字段为空也显示右左表内容,左右边字段补NULL
SELECT * FROM table_name1 LEFT JOIN table_name2 ON table_name1.column_name = table_name2.column_name;
SELECT * FROM table_name1 RIGHT JOIN table_name2 ON table_name1.column_name = table_name2.column_name; 

-- 子查询 
SELECT * FROM table_name WHERE column_name IN (SELECT column_name FROM table_name ...);
SELECT * FROM table_name WHERE column_name > ALL(SELECT column_name FROM table_name ...);
SELECT * FROM table_name WHERE column_name = ANY(SELECT column_name FROM table_name ...);

-- 事务 ACID (原子性, 一致性, 隔离性, 持久性)
START TRANSACTION; -- 开启事务
COMMIT; -- 提交/结束事务,把事务中所有修改永久保存到数据库
-- 切记COMMIT之后就无法再回滚  
ROLLBACK; -- 回滚事务,撤销事务中修改
SAVEPOINT savepoint_name; -- 设置保存点
ROLLBACK TO savepoint_name; -- 回滚到保存点
RELEASE SAVEPOINT savepoint_name; -- 释放保存点

-- 数据类型 
-- tinyint 1字节 smallint 2字节 mediumint 3字节 int 4字节 bigint 8字节
-- float(alldigitsN, n)后7位 double(alldigitsN, n)后15位 decimal(m, n)
