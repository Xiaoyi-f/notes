WSL 的全称是 Windows Subsystem for Linux（适用于 Linux 的 Windows 子系统）
终端输入命令 
wsl --install // 自动安装适配版本 默认下载在C盘
wsl --set-default-version 2 // 设置wsl版本为2
wsl --update --web-download  // 更新wsl

# wsl 迁移 
wsl --list --verbose 
wsl --export <distribution> <file.tar>
wsl --unregister <distribution> 

# tar 会完整保存 Linux 文件元数据
# 导入后需要重新设置默认用户，否则默认 root 登录
wsl --import <distribution> <path> <file.tar> --version 2

## linux 
文件系统
/bin 基础命令
/sbin 系统管理命令
/lib /lib64 系统库文件 
/etc 系统配置文件
/var 动态杂物间 -> 系统日志 ... 
/tmp 临时仓库 重启清空 
/home 普通用户的家 普通用户登录默认在此目录下
/root root用户的家 普通用户进不了 
/dev 设备文件
/proc /sys 运行文件库 
/usr 用户软件仓库 apt install 软件大多在这里 
/opt 可选第三方软件 通常存放手动安装的大型闭源软件/独立软件包 
/mnt 挂载点 存放已挂载的磁盘文件

# 核心命令
sudo -i // 切换到root用户
\ // 反斜杠续行符 
mkdir -p /dir/dir // 创建多层目录 
rm -rf 
ls -la
