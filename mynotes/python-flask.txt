表名默认就是类名小写 如果是驼峰命名的话，默认就是使用_(下划线隔离)
class Table(db.Model):
    """
    field = db.Column(db.Integer, primary_key=True, autoincrement=True)
    ->
        db.String(n) unique=True nullable=False
        db.ForeignKey('table.field') default=val
        db.Float db.Numeric(整体位数, 小数位数) db.Boolean db.JSON(使用copy.deepcopy操作)
        数组和字典都属于json
    """

    def toDict(self):
        data = {
            "xx": self.xx 
        }
        
        return data
