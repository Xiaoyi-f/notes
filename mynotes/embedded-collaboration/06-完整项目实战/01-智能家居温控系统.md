# 项目实战1：智能家居温控系统——从零到完整系统

---

## 一、项目需求

```
系统功能：
1. 每个房间部署温湿度传感器（ESP32）
2. 手机App实时查看各房间温度
3. 温度超过设定值自动告警
4. 可远程控制空调/风扇开关
5. 历史温度曲线查看
6. 设备OTA升级

硬件清单：
- ESP32开发板 × 5（客厅、卧室、厨房、书房、阳台）
- DHT22温湿度传感器 × 5
- 继电器模块 × 3（控制客厅/卧室/书房空调）
- 总成本：约 ¥200
```

## 三、通信协议设计

```
═══════════════════════════════════════════════════════════════
           智能家居温控系统通信协议 V1.0
═══════════════════════════════════════════════════════════════

【通信方式】
- 设备→云端：MQTT
- 云端→设备：MQTT
- App→云端：HTTPS REST API

【MQTT Topic设计】
设备上报：
  home/{room_id}/sensor/temperature    ← 温度（retain=true）
  home/{room_id}/sensor/humidity       ← 湿度（retain=true）
  home/{room_id}/device/heartbeat      ← 心跳
  home/{room_id}/device/status         ← 设备状态

云端下发：
  home/{room_id}/control/ac            ← 空调控制
  home/{room_id}/config/threshold      ← 阈值配置
  home/{room_id}/ota/cmd               ← OTA升级

【消息格式（JSON）】
温度上报：
  {
    "value": 25.5,       // 温度值
    "unit": "celsius",   // 单位
    "timestamp": 1700000000
  }

空调控制：
  {
    "action": "on",      // on/off
    "temperature": 26,   // 设定温度
    "mode": "cool"       // cool/heat/fan
  }

心跳：
  {
    "type": "heartbeat",
    "timestamp": 1700000000,
    "status": {
      "wifiRssi": -65,
      "freeHeap": 120000,
      "uptime": 3600
    }
  }
═══════════════════════════════════════════════════════════════
```

## 四、嵌入式端代码框架（ESP32/Arduino）

```cpp
// device_main.cpp
#include <WiFi.h>
#include <PubSubClient.h>
#include <DHT.h>
#include <ArduinoJson.h>

// ========== 配置 ==========
const char* WIFI_SSID = "your_wifi";
const char* WIFI_PASS = "your_password";
const char* MQTT_BROKER = "broker.emqx.io";
const int MQTT_PORT = 1883;
const char* ROOM_ID = "livingroom";  // 每个设备改这里
const int DHT_PIN = 4;

// ========== 全局变量 ==========
WiFiClient wifiClient;
PubSubClient mqtt(wifiClient);
DHT dht(DHT_PIN, DHT22);

float tempThreshold = 30.0;  // 告警阈值
bool acOn = false;
unsigned long lastReport = 0;
unsigned long lastHeartbeat = 0;

// ========== 连接WiFi ==========
void connectWiFi() {
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi连接成功");
}

// ========== MQTT回调 ==========
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }

  // 解析JSON
  StaticJsonDocument<256> doc;
  deserializeJson(doc, message);

  String topicStr = String(topic);

  // 处理空调控制
  if (topicStr.endsWith("/control/ac")) {
    const char* action = doc["action"];
    if (strcmp(action, "on") == 0) {
      digitalWrite(5, HIGH);  // 开继电器
      acOn = true;
    } else {
      digitalWrite(5, LOW);
      acOn = false;
    }
    reportStatus();
  }

  // 处理阈值配置
  if (topicStr.endsWith("/config/threshold")) {
    tempThreshold = doc["temperature"];
  }
}

// ========== 连接MQTT ==========
void connectMQTT() {
  String clientId = "home_" + String(ROOM_ID) + "_" + String(random(0xffff), HEX);

  while (!mqtt.connected()) {
    if (mqtt.connect(clientId.c_str())) {
      Serial.println("MQTT连接成功");

      // 订阅控制Topic
      mqtt.subscribe(("home/" + String(ROOM_ID) + "/control/ac").c_str());
      mqtt.subscribe(("home/" + String(ROOM_ID) + "/config/threshold").c_str());
    } else {
      delay(5000);
    }
  }
}

// ========== 上报温度 ==========
void reportTemperature() {
  float temp = dht.readTemperature();
  float humi = dht.readHumidity();

  if (isnan(temp) || isnan(humi)) return;

  StaticJsonDocument<256> doc;
  doc["value"] = temp;
  doc["unit"] = "celsius";
  doc["timestamp"] = millis() / 1000;

  char buffer[256];
  serializeJson(doc, buffer);

  String topic = "home/" + String(ROOM_ID) + "/sensor/temperature";
  mqtt.publish(topic.c_str(), buffer, true);  // retain=true

  // 检查告警
  if (temp > tempThreshold && !acOn) {
    StaticJsonDocument<256> alarm;
    alarm["type"] = "high_temperature";
    alarm["value"] = temp;
    alarm["threshold"] = tempThreshold;
    serializeJson(alarm, buffer);
    mqtt.publish("home/alarm", buffer);
  }
}

// ========== 上报心跳 ==========
void reportHeartbeat() {
  StaticJsonDocument<256> doc;
  doc["type"] = "heartbeat";
  doc["timestamp"] = millis() / 1000;
  doc["status"]["wifiRssi"] = WiFi.RSSI();
  doc["status"]["freeHeap"] = ESP.getFreeHeap();
  doc["status"]["uptime"] = millis() / 1000;

  char buffer[256];
  serializeJson(doc, buffer);

  String topic = "home/" + String(ROOM_ID) + "/device/heartbeat";
  mqtt.publish(topic.c_str(), buffer);
}

// ========== 主循环 ==========
void setup() {
  Serial.begin(115200);
  dht.begin();
  pinMode(5, OUTPUT);  // 继电器引脚

  connectWiFi();

  mqtt.setServer(MQTT_BROKER, MQTT_PORT);
  mqtt.setCallback(mqttCallback);
  connectMQTT();
}

void loop() {
  if (!mqtt.connected()) {
    connectMQTT();
  }
  mqtt.loop();

  unsigned long now = millis();

  // 每5秒上报温度
  if (now - lastReport > 5000) {
    reportTemperature();
    lastReport = now;
  }

  // 每30秒心跳
  if (now - lastHeartbeat > 30000) {
    reportHeartbeat();
    lastHeartbeat = now;
  }
}
```

## 五、UniApp 端代码框架

```vue
<!-- pages/home/home.vue -->
<template>
  <view class="home">
    <view v-for="room in rooms" :key="room.id" class="room-card">
      <view class="room-name">{{ room.name }}</view>
      <view class="temp-display">
        <text class="temp-value">{{ room.temperature }}</text>
        <text class="temp-unit">°C</text>
      </view>
      <view class="humidity">湿度 {{ room.humidity }}%</view>

      <view class="ac-control">
        <switch
          :checked="room.acOn"
          @change="toggleAC(room.id, $event.detail.value)"
        />
        <text>{{ room.acOn ? '空调开启' : '空调关闭' }}</text>
      </view>

      <view class="threshold">
        告警阈值：{{ room.threshold }}°C
        <slider
          :value="room.threshold"
          min="20" max="40"
          @change="setThreshold(room.id, $event.detail.value)"
        />
      </view>
    </view>

    <button @click="goToHistory">查看历史曲线</button>
  </view>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import mqtt from 'mqtt/dist/mqtt.esm'

interface Room {
  id: string
  name: string
  temperature: number
  humidity: number
  acOn: boolean
  threshold: number
  online: boolean
}

const rooms = ref<Room[]>([
  { id: 'livingroom', name: '客厅', temperature: 0, humidity: 0, acOn: false, threshold: 30, online: false },
  { id: 'bedroom', name: '卧室', temperature: 0, humidity: 0, acOn: false, threshold: 28, online: false },
  { id: 'kitchen', name: '厨房', temperature: 0, humidity: 0, acOn: false, threshold: 35, online: false }
])

let mqttClient: any = null

onMounted(() => {
  // 连接MQTT
  mqttClient = mqtt.connect('wss://broker.emqx.io:8084/mqtt')

  mqttClient.on('connect', () => {
    // 订阅所有房间的温度
    rooms.value.forEach(room => {
      mqttClient.subscribe(`home/${room.id}/sensor/temperature`)
      mqttClient.subscribe(`home/${room.id}/device/status`)
    })
  })

  mqttClient.on('message', (topic: string, message: Buffer) => {
    const data = JSON.parse(message.toString())
    const roomId = topic.split('/')[1]
    const room = rooms.value.find(r => r.id === roomId)

    if (!room) return

    if (topic.includes('temperature')) {
      room.temperature = data.value
    }
    if (topic.includes('status')) {
      room.acOn = data.acOn
    }
  })
})

const toggleAC = (roomId: string, on: boolean) => {
  mqttClient.publish(`home/${roomId}/control/ac`, JSON.stringify({
    action: on ? 'on' : 'off',
    temperature: 26,
    mode: 'cool'
  }))
}

const setThreshold = (roomId: string, value: number) => {
  mqttClient.publish(`home/${roomId}/config/threshold`, JSON.stringify({
    temperature: value
  }))
}

const goToHistory = () => {
  uni.navigateTo({ url: '/pages/history/history' })
}
</script>
```
