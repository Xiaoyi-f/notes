## 二、安全最佳实践

### 1. 预加载脚本安全暴露

```javascript
// electron/preload.js
const { contextBridge, ipcRenderer } = require('electron')

// 白名单通道
const VALID_SEND_CHANNELS = ['toMain', 'window-control', 'file-operation']
const VALID_RECEIVE_CHANNELS = ['fromMain', 'download-progress', 'update-available']
const VALID_INVOKE_CHANNELS = ['get-version', 'open-dialog', 'save-file']

contextBridge.exposeInMainWorld('electronAPI', {
  // 发送消息（单向）
  send: (channel, data) => {
    if (VALID_SEND_CHANNELS.includes(channel)) {
      ipcRenderer.send(channel, data)
    }
  },
  
  // 接收消息
  on: (channel, callback) => {
    if (VALID_RECEIVE_CHANNELS.includes(channel)) {
      const subscription = (event, ...args) => callback(...args)
      ipcRenderer.on(channel, subscription)
      
      // 返回取消订阅函数
      return () => ipcRenderer.removeListener(channel, subscription)
    }
  },
  
  // 调用并等待返回
  invoke: (channel, data) => {
    if (VALID_INVOKE_CHANNELS.includes(channel)) {
      return ipcRenderer.invoke(channel, data)
    }
    return Promise.reject(new Error(`Invalid channel: ${channel}`))
  }
})
```

## 三、IPC通信模式

### 模式一：单向发送（fire-and-forget）

```javascript
// 渲染进程 -> 主进程
window.electronAPI.send('toMain', { action: 'doSomething' })

// 主进程接收
ipcMain.on('toMain', (event, data) => {
  console.log('收到:', data)
})
```

### 模式二：请求-响应（invoke/handle）

```javascript
// 渲染进程调用
const result = await window.electronAPI.invoke('get-version')
console.log(result)

// 主进程处理
ipcMain.handle('get-version', () => {
  return app.getVersion()
})
```

### 模式三：主进程主动推送

```javascript
// 主进程发送
mainWindow.webContents.send('update-available', { version: '1.0.1' })

// 渲染进程接收
window.electronAPI.on('update-available', (data) => {
  console.log('有新版本:', data.version)
})
```

## 四、组合式函数封装

```javascript
// src/composables/useElectron.js
import { ref, onMounted, onUnmounted } from 'vue'

export function useElectron() {
  const api = window.electronAPI
  
  if (!api) {
    console.warn('Electron API not available')
    return { isElectron: ref(false) }
  }
  
  return {
    isElectron: ref(true),
    api
  }
}

// src/composables/useTheme.js
import { ref, watch } from 'vue'

export function useTheme() {
  const isDark = ref(localStorage.getItem('theme') === 'dark')
  
  watch(isDark, (val) => {
    localStorage.setItem('theme', val ? 'dark' : 'light')
    document.documentElement.setAttribute('data-theme', val ? 'dark' : 'light')
  }, { immediate: true })
  
  const toggleTheme = () => {
    isDark.value = !isDark.value
  }
  
  return { isDark, toggleTheme }
}

// src/composables/useNotify.js
export function useNotify() {
  const notify = async (options) => {
    if (window.electronAPI?.notification) {
      await window.electronAPI.notification.show(options)
    } else {
      // 降级到Web Notification
      new Notification(options.title, {
        body: options.body,
        icon: options.icon
      })
    }
  }
  
  return { notify }
}
```

## 五、开发规范

### 命名规范

| 类型 | 规范 | 示例 |
|------|------|------|
| IPC通道 | 小写+连字符 | `window-minimize`, `file-read` |
| 窗口名称 | 大驼峰 | `MainWindow`, `LoginWindow` |
| 组件 | 大驼峰 | `AppHeader.vue`, `UserPanel.vue` |
| 组合式函数 | use+大驼峰 | `useTheme.js`, `useElectron.js` |
| 常量 | 大写+下划线 | `DEFAULT_WINDOW_WIDTH` |

### 代码组织

```javascript
// electron/utils/ipcHandlers.js
const { ipcMain, dialog, shell } = require('electron')
const fs = require('fs')

// 统一注册所有IPC处理器
function registerIpcHandlers(mainWindow) {
  
  // ========== 文件操作 ==========
  ipcMain.handle('file-read', async (event, filePath) => {
    // ...
  })
  
  ipcMain.handle('file-write', async (event, { filePath, content }) => {
    // ...
  })
  
  // ========== 窗口操作 ==========
  ipcMain.on('window-minimize', () => {
    mainWindow.minimize()
  })
  
  ipcMain.on('window-maximize', () => {
    if (mainWindow.isMaximized()) {
      mainWindow.unmaximize()
    } else {
      mainWindow.maximize()
    }
  })
  
  // ========== 系统操作 ==========
  ipcMain.handle('open-external', (event, url) => {
    shell.openExternal(url)
  })
}

module.exports = { registerIpcHandlers }
```

### 主进程入口组织

```javascript
// electron/main.js
const { app, BrowserWindow } = require('electron')
const path = require('path')
const { registerIpcHandlers } = require('./utils/ipcHandlers')
const { createMenu } = require('./utils/menu')
const { createTray } = require('./utils/tray')
const { setupAutoUpdater } = require('./utils/autoUpdate')

let mainWindow

function createMainWindow() {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false
    }
  })
  
  // 加载页面
  if (process.env.VITE_DEV_SERVER_URL) {
    mainWindow.loadURL(process.env.VITE_DEV_SERVER_URL)
    mainWindow.webContents.openDevTools()
  } else {
    mainWindow.loadFile(path.join(__dirname, '../dist/index.html'))
  }
  
  return mainWindow
}

app.whenReady().then(() => {
  const win = createMainWindow()
  
  // 注册IPC
  registerIpcHandlers(win)
  
  // 创建菜单
  createMenu(win)
  
  // 创建托盘
  createTray(win)
  
  // 设置自动更新
  setupAutoUpdater(win)
})

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') app.quit()
})

app.on('activate', () => {
  if (BrowserWindow.getAllWindows().length === 0) createMainWindow()
})
```
