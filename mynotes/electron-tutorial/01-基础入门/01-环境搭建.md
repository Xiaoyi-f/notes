```javascript
// main.js - 主进程入口
const { app, BrowserWindow } = require('electron')
const path = require('path')

// 创建窗口的函数
function createWindow() {
  // 创建浏览器窗口
  const mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      // 预加载脚本路径
      preload: path.join(__dirname, 'preload.js'),
      // 安全设置
      contextIsolation: true,    // 启用上下文隔离（安全）
      nodeIntegration: false,    // 禁用Node集成（安全）
      sandbox: true              // 启用沙箱（安全）
    }
  })

  // 加载页面
  // 开发环境加载本地文件
  mainWindow.loadFile('index.html')
  
  // 开发时打开开发者工具
  mainWindow.webContents.openDevTools()
}

// Electron 初始化完成后创建窗口
app.whenReady().then(() => {
  createWindow()

  // macOS: 点击dock图标时重新创建窗口
  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) {
      createWindow()
    }
  })
})

// 所有窗口关闭时退出应用（Windows/Linux）
app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit()
  }
})
```

```javascript
// preload.js - 预加载脚本（安全桥梁）
const { contextBridge, ipcRenderer } = require('electron')

// 暴露安全的API给前端
contextBridge.exposeInMainWorld('electronAPI', {
  // 发送消息到主进程
  sendMessage: (channel, data) => {
    // 白名单，只允许特定通道
    const validChannels = ['toMain']
    if (validChannels.includes(channel)) {
      ipcRenderer.send(channel, data)
    }
  },
  
  // 接收主进程消息
  onMessage: (channel, callback) => {
    const validChannels = ['fromMain']
    if (validChannels.includes(channel)) {
      ipcRenderer.on(channel, (event, ...args) => callback(...args))
    }
  },
  
  // 调用主进程并等待返回（invoke/handle模式）
  invoke: (channel, data) => {
    const validChannels = ['get-app-version', 'open-file']
    if (validChannels.includes(channel)) {
      return ipcRenderer.invoke(channel, data)
    }
  }
})
```

## 五、核心模块速查

### app - 应用生命周期

```javascript
const { app } = require('electron')

app.whenReady()           // 应用初始化完成
app.on('ready', fn)       // 准备就绪
app.on('window-all-closed', fn)  // 所有窗口关闭
app.on('before-quit', fn) // 退出前
app.on('will-quit', fn)   // 即将退出
app.on('quit', fn)        // 已退出
app.on('activate', fn)    // macOS dock点击（无窗口时）

app.getVersion()          // 获取版本号
app.getPath('userData')   // 获取用户数据目录
app.setName('AppName')    // 设置应用名称
```

### BrowserWindow - 窗口管理

```javascript
const { BrowserWindow } = require('electron')

const win = new BrowserWindow({
  width: 1200,
  height: 800,
  minWidth: 800,          // 最小宽度
  minHeight: 600,         // 最小高度
  x: 100,                 // 窗口位置
  y: 100,
  center: true,           // 居中显示
  resizable: true,        // 可调整大小
  maximizable: true,      // 可最大化
  minimizable: true,      // 可最小化
  closable: true,         // 可关闭
  fullscreenable: true,   // 可全屏
  alwaysOnTop: false,     // 置顶
  show: false,            // 创建时不显示（加载完再显示）
  frame: true,            // 显示边框和标题栏（false=无边框窗口）
  transparent: false,     // 透明窗口
  opacity: 1,             // 透明度
  backgroundColor: '#fff',// 背景色
  icon: 'path/to/icon.png',// 窗口图标
  title: '窗口标题',
  webPreferences: {
    preload: path.join(__dirname, 'preload.js'),
    contextIsolation: true,
    nodeIntegration: false,
    sandbox: true,
    webSecurity: true,    // 同源策略
    allowRunningInsecureContent: false
  }
})

// 窗口方法
win.loadURL('https://example.com')     // 加载URL
win.loadFile('index.html')             // 加载本地文件
win.show()                             // 显示窗口
win.hide()                             // 隐藏窗口
win.close()                            // 关闭窗口
win.focus()                            // 聚焦窗口
win.blur()                             // 失焦
win.maximize()                         // 最大化
win.unmaximize()                       // 取消最大化
win.minimize()                         // 最小化
win.restore()                          // 恢复
win.setFullScreen(true)                // 全屏
win.setSize(width, height)             // 设置大小
win.setPosition(x, y)                  // 设置位置
win.webContents.openDevTools()         // 打开开发者工具
win.webContents.send('channel', data)  // 向渲染进程发送消息
```
