## 一、文件系统操作

### 1. 读取文件

```javascript
// main.js - 主进程中处理文件操作
const { ipcMain, dialog } = require('electron')
const fs = require('fs')
const path = require('path')

// 读取文件内容
ipcMain.handle('file-read', async (event, filePath) => {
  try {
    const content = fs.readFileSync(filePath, 'utf-8')
    return { success: true, content }
  } catch (err) {
    return { success: false, error: err.message }
  }
})

// 写入文件
ipcMain.handle('file-write', async (event, { filePath, content }) => {
  try {
    fs.writeFileSync(filePath, content, 'utf-8')
    return { success: true }
  } catch (err) {
    return { success: false, error: err.message }
  }
})

// 选择文件并读取
ipcMain.handle('file-select-and-read', async () => {
  const result = await dialog.showOpenDialog({
    properties: ['openFile'],
    filters: [
      { name: '文本文件', extensions: ['txt', 'md', 'json'] },
      { name: '所有文件', extensions: ['*'] }
    ]
  })
  
  if (result.canceled) return { canceled: true }
  
  const filePath = result.filePaths[0]
  const content = fs.readFileSync(filePath, 'utf-8')
  const stats = fs.statSync(filePath)
  
  return {
    canceled: false,
    filePath,
    fileName: path.basename(filePath),
    content,
    size: stats.size,
    modifiedTime: stats.mtime
  }
})
```

```javascript
// preload.js
const { contextBridge, ipcRenderer } = require('electron')

contextBridge.exposeInMainWorld('electronAPI', {
  file: {
    read: (filePath) => ipcRenderer.invoke('file-read', filePath),
    write: (filePath, content) => ipcRenderer.invoke('file-write', { filePath, content }),
    selectAndRead: () => ipcRenderer.invoke('file-select-and-read')
  }
})
```

```vue
<!-- 前端使用 -->
<template>
  <div>
    <button @click="selectFile">选择文件</button>
    <div v-if="fileInfo">
      <p>文件名：{{ fileInfo.fileName }}</p>
      <p>大小：{{ formatSize(fileInfo.size) }}</p>
      <textarea v-model="fileInfo.content" rows="20" cols="80"></textarea>
      <button @click="saveFile">保存</button>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'

const fileInfo = ref(null)

const selectFile = async () => {
  const result = await window.electronAPI.file.selectAndRead()
  if (!result.canceled) {
    fileInfo.value = result
  }
}

const saveFile = async () => {
  const result = await window.electronAPI.file.write(
    fileInfo.value.filePath,
    fileInfo.value.content
  )
  if (result.success) {
    alert('保存成功')
  }
}

const formatSize = (bytes) => {
  if (bytes < 1024) return bytes + ' B'
  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(2) + ' KB'
  return (bytes / 1024 / 1024).toFixed(2) + ' MB'
}
</script>
```

### 2. 文件拖放

```vue
<template>
  <div
    class="drop-zone"
    @dragover.prevent
    @drop="onDrop"
    :class="{ 'drag-over': isDragOver }"
    @dragenter="isDragOver = true"
    @dragleave="isDragOver = false"
  >
    <p>拖拽文件到这里</p>
  </div>
</template>

<script setup>
import { ref } from 'vue'

const isDragOver = ref(false)

const onDrop = async (e) => {
  e.preventDefault()
  isDragOver.value = false
  
  const files = e.dataTransfer.files
  for (const file of files) {
    // 读取文件内容
    const content = await window.electronAPI.file.read(file.path)
    console.log(file.name, content)
  }
}
</script>
```

## 二、目录操作

```javascript
// main.js
const fs = require('fs')
const path = require('path')
const os = require('os')

// 获取特殊目录路径
ipcMain.handle('path-get', (event, name) => {
  // name: home, appData, userData, temp, desktop, documents, downloads, music, pictures, videos
  return app.getPath(name)
})

// 读取目录
ipcMain.handle('dir-read', (event, dirPath) => {
  try {
    const items = fs.readdirSync(dirPath, { withFileTypes: true })
    return items.map(item => ({
      name: item.name,
      isDirectory: item.isDirectory(),
      isFile: item.isFile(),
      path: path.join(dirPath, item.name)
    }))
  } catch (err) {
    return { error: err.message }
  }
})

// 创建目录
ipcMain.handle('dir-create', (event, dirPath) => {
  try {
    fs.mkdirSync(dirPath, { recursive: true })
    return { success: true }
  } catch (err) {
    return { success: false, error: err.message }
  }
})

// 删除文件/目录
ipcMain.handle('file-delete', (event, filePath) => {
  try {
    const stat = fs.statSync(filePath)
    if (stat.isDirectory()) {
      fs.rmdirSync(filePath, { recursive: true })
    } else {
      fs.unlinkSync(filePath)
    }
    return { success: true }
  } catch (err) {
    return { success: false, error: err.message }
  }
})

// 监听文件变化
ipcMain.handle('file-watch', (event, filePath) => {
  const watcher = fs.watch(filePath, (eventType, filename) => {
    // 发送变化事件到渲染进程
    const win = BrowserWindow.fromWebContents(event.sender)
    win.webContents.send('file-changed', { eventType, filename })
  })
  
  return watcher
})
```

## 三、剪贴板操作

```javascript
// main.js
const { clipboard, ipcMain } = require('electron')

ipcMain.handle('clipboard-read-text', () => {
  return clipboard.readText()
})

ipcMain.handle('clipboard-write-text', (event, text) => {
  clipboard.writeText(text)
})

ipcMain.handle('clipboard-read-image', () => {
  const image = clipboard.readImage()
  if (image.isEmpty()) return null
  return image.toDataURL()  // 返回base64
})

ipcMain.handle('clipboard-write-image', (event, imagePath) => {
  const image = nativeImage.createFromPath(imagePath)
  clipboard.writeImage(image)
})

// 读取HTML
ipcMain.handle('clipboard-read-html', () => {
  return clipboard.readHTML()
})

// 读取RTF
ipcMain.handle('clipboard-read-rtf', () => {
  return clipboard.readRTF()
})

// 清空剪贴板
ipcMain.handle('clipboard-clear', () => {
  clipboard.clear()
})
```

```javascript
// preload.js
contextBridge.exposeInMainWorld('electronAPI', {

    writeText: (text) => ipcRenderer.invoke('clipboard-write-text', text),
    readImage: () => ipcRenderer.invoke('clipboard-read-image'),
    writeImage: (path) => ipcRenderer.invoke('clipboard-write-image', path),
    clear: () => ipcRenderer.invoke('clipboard-clear')
  }
})
```


## 四、本地数据持久化

### 使用 electron-store（推荐）

```bash
npm install electron-store
```

```javascript
// main.js
const Store = require('electron-store')

// 创建配置存储
const store = new Store({
  name: 'app-config',  // 配置文件名
  defaults: {
    windowBounds: { width: 1200, height: 800 },
    userSettings: {
      theme: 'light',
      language: 'zh-CN',
      autoStart: false
    },
    recentFiles: []
  }
})

// 保存窗口大小
mainWindow.on('resize', () => {
  const { width, height } = mainWindow.getBounds()
  store.set('windowBounds', { width, height })
})

// IPC接口
ipcMain.handle('store-get', (event, key) => store.get(key))
ipcMain.handle('store-set', (event, key, value) => store.set(key, value))
ipcMain.handle('store-delete', (event, key) => store.delete(key))
ipcMain.handle('store-clear', () => store.clear())
```

```javascript
// preload.js
contextBridge.exposeInMainWorld('electronAPI', {
  store: {
    get: (key) => ipcRenderer.invoke('store-get', key),
    set: (key, value) => ipcRenderer.invoke('store-set', key, value),
    delete: (key) => ipcRenderer.invoke('store-delete', key),
    clear: () => ipcRenderer.invoke('store-clear')
  }
})
```

```vue
<!-- 设置页面 -->
<script setup>
import { ref, onMounted } from 'vue'

const settings = ref({
  theme: 'light',
  language: 'zh-CN',
  autoStart: false
})

  settings.value = await window.electronAPI.store.get('userSettings')
})

  await window.electronAPI.store.set('userSettings', settings.value)
  alert('设置已保存')
}
</script>
```

## 五、系统通知

```javascript
// main.js
const { Notification } = require('electron')

// 发送系统通知
ipcMain.handle('notification-show', (event, options) => {
  const notification = new Notification({
    title: options.title || '通知',
    body: options.body || '',
    icon: path.join(__dirname, 'assets', 'icon.png'),
    silent: false,           // 是否有声音
    timeoutType: 'default'   // default/never
  })
  
  notification.on('click', () => {
    // 点击通知时聚焦窗口
    const win = BrowserWindow.getAllWindows()[0]
    win?.show()
    win?.focus()
  })
  
  notification.show()
})

// 检查通知权限
ipcMain.handle('notification-check', () => {
  return Notification.isSupported()
})
```

```vue
<script setup>
const showNotification = async () => {
  await window.electronAPI.notification.show({
    title: '任务完成',
    body: '文件下载已完成'
  })
}
</script>
```
