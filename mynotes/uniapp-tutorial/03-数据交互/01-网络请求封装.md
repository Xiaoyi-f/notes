# 01 - 网络请求封装（连接后端API）

> UniApp 通过 `uni.request` 发起HTTP请求，连接部署在云服务器上的后端API。

---

## 一、uni.request 基础用法

```javascript
// 最基础的请求
uni.request({
  url: 'https://你的域名/api/users',
  method: 'GET',
  success: (res) => {
    console.log('成功：', res.data)
  },
  fail: (err) => {
    console.log('失败：', err)
  }
})
```

## 二、封装请求工具（项目必备）

### 1. 创建 request.js

```javascript
// utils/request.js

// 基础配置
const BASE_URL = 'https://api.yoursite.com'  // 你的云服务器域名
// 开发环境可以用：const BASE_URL = 'http://localhost:8080'

// 请求拦截器
const request = (options) => {
  return new Promise((resolve, reject) => {
    
    // 显示加载中（可选）
    if (options.loading !== false) {
      uni.showLoading({ title: '加载中...', mask: true })
    }
    
    // 从本地存储获取token
    const token = uni.getStorageSync('token')
    
    uni.request({
      url: BASE_URL + options.url,
      method: options.method || 'GET',
      data: options.data || {},
      header: {
        'Content-Type': 'application/json',
        'Authorization': token ? `Bearer ${token}` : ''
      },
      timeout: 30000,  // 30秒超时
      
      success: (res) => {
        uni.hideLoading()
        
        if (res.statusCode === 200) {
          // 后端返回正常
          if (res.data.code === 200 || res.data.code === 0) {
            resolve(res.data)
          } else {
            // 业务错误（如参数错误、权限不足等）
            uni.showToast({ 
              title: res.data.msg || '请求失败', 
              icon: 'none' 
            })
            reject(res.data)
          }
        } else if (res.statusCode === 401) {
          // Token过期，跳转登录
          uni.showToast({ title: '登录已过期', icon: 'none' })
          uni.removeStorageSync('token')
          uni.navigateTo({ url: '/pages/login/login' })
          reject(res)
        } else {
          // HTTP错误
          uni.showToast({ 
            title: `服务器错误：${res.statusCode}`, 
            icon: 'none' 
          })
          reject(res)
        }
      },
      
      fail: (err) => {
        uni.hideLoading()
        uni.showToast({ 
          title: '网络连接失败，请检查网络', 
          icon: 'none' 
        })
        reject(err)
      }
    })
  })
}

// 导出常用方法
export default {
  get(url, params = {}, options = {}) {
    return request({ url, method: 'GET', data: params, ...options })
  },
  
  post(url, data = {}, options = {}) {
    return request({ url, method: 'POST', data, ...options })
  },
  
  put(url, data = {}, options = {}) {
    return request({ url, method: 'PUT', data, ...options })
  },
  
  delete(url, params = {}, options = {}) {
    return request({ url, method: 'DELETE', data: params, ...options })
  },
  
  // 上传文件
  upload(url, filePath, formData = {}) {
    return new Promise((resolve, reject) => {
      const token = uni.getStorageSync('token')
      
      uni.uploadFile({
        url: BASE_URL + url,
        filePath,
        name: 'file',
        formData,
        header: {
          'Authorization': token ? `Bearer ${token}` : ''
        },
        success: (res) => {
          const data = JSON.parse(res.data)
          resolve(data)
        },
        fail: reject
      })
    })
  }
}
```

### 2. 创建 API 接口管理文件

```javascript
// api/index.js
import http from '@/utils/request.js'

// 用户相关
export const userApi = {
  // 登录
  login: (data) => http.post('/api/user/login', data),
  
  // 注册
  register: (data) => http.post('/api/user/register', data),
  
  // 获取用户信息
  getUserInfo: () => http.get('/api/user/info'),
  
  // 修改用户信息
  updateUserInfo: (data) => http.put('/api/user/info', data),
  
  // 修改头像
  updateAvatar: (filePath) => http.upload('/api/user/avatar', filePath)
}

// 商品相关
export const goodsApi = {
  // 商品列表
  getList: (params) => http.get('/api/goods/list', params),
  
  // 商品详情
  getDetail: (id) => http.get(`/api/goods/detail/${id}`),
  
  // 搜索商品
  search: (keyword) => http.get('/api/goods/search', { keyword })
}

// 订单相关
export const orderApi = {
  // 创建订单
  create: (data) => http.post('/api/order/create', data),
  
  // 订单列表
  getList: (params) => http.get('/api/order/list', params),
  
  // 订单详情
  getDetail: (id) => http.get(`/api/order/detail/${id}`),
  
  // 取消订单
  cancel: (id) => http.put(`/api/order/cancel/${id}`)
}
```

### 3. 在页面中使用

```vue
<script setup>
import { ref, onLoad } from '@dcloudio/uni-app'
import { userApi, goodsApi } from '@/api/index.js'

const userInfo = ref({})
const goodsList = ref([])
const loading = ref(false)

// 页面加载时获取数据
onLoad(async () => {
  await getUserInfo()
  await getGoodsList()
})

// 获取用户信息
const getUserInfo = async () => {
  try {
    const res = await userApi.getUserInfo()
    userInfo.value = res.data
  } catch (err) {
    console.log('获取用户信息失败', err)
  }
}

// 获取商品列表
const getGoodsList = async () => {
  loading.value = true
  try {
    const res = await goodsApi.getList({ 
      page: 1, 
      size: 10 
    })
    goodsList.value = res.data.list
  } catch (err) {
    console.log('获取商品列表失败', err)
  } finally {
    loading.value = false
  }
}

// 登录
const handleLogin = async () => {
  try {
    const res = await userApi.login({
      username: 'test',
      password: '123456'
    })
    
    // 保存token
    uni.setStorageSync('token', res.data.token)
    
    uni.showToast({ title: '登录成功', icon: 'success' })
    uni.switchTab({ url: '/pages/index/index' })
  } catch (err) {
    console.log('登录失败', err)
  }
}

// 上传头像
const chooseAvatar = () => {
  uni.chooseImage({
    count: 1,
    success: async (res) => {
      const filePath = res.tempFilePaths[0]
      try {
        const result = await userApi.updateAvatar(filePath)
        userInfo.value.avatar = result.data.url
        uni.showToast({ title: '上传成功' })
      } catch (err) {
        uni.showToast({ title: '上传失败', icon: 'none' })
      }
    }
  })
}
</script>
```

## 三、分页加载（列表必备）

```vue
<script setup>
import { ref, onLoad, onReachBottom, onPullDownRefresh } from '@dcloudio/uni-app'
import { goodsApi } from '@/api/index.js'

const list = ref([])
const page = ref(1)
const size = ref(10)
const total = ref(0)
const loading = ref(false)
const finished = ref(false)

// 获取列表
const fetchList = async (isRefresh = false) => {
  if (loading.value) return
  if (finished.value && !isRefresh) return
  
  loading.value = true
  
  try {
    const res = await goodsApi.getList({
      page: isRefresh ? 1 : page.value,
      size: size.value
    })
    
    if (isRefresh) {
      list.value = res.data.list
      page.value = 2
      finished.value = false
      uni.stopPullDownRefresh()
    } else {
      list.value.push(...res.data.list)
      page.value++
    }
    
    total.value = res.data.total
    
    // 判断是否加载完毕
    if (list.value.length >= total.value) {
      finished.value = true
    }
  } catch (err) {
    console.log('加载失败', err)
  } finally {
    loading.value = false
  }
}

// 下拉刷新
onPullDownRefresh(() => {
  fetchList(true)
})

// 上拉加载更多
onReachBottom(() => {
  fetchList(false)
})

onLoad(() => {
  fetchList(true)
})
</script>

<template>
  <view>
    <view v-for="item in list" :key="item.id" class="item">
      {{ item.name }}
    </view>
    
    <view class="status">
      <text v-if="loading">加载中...</text>
      <text v-else-if="finished">没有更多了</text>
    </view>
  </view>
</template>

<style>
.item { padding: 30rpx; border-bottom: 1rpx solid #eee; }
.status { text-align: center; padding: 30rpx; color: #999; }
</style>
```

## 四、完整登录页示例

```vue
<!-- pages/login/login.vue -->
<template>
  <view class="login-page">
    <view class="logo">
      <image src="/static/logo.png" class="logo-img" />
      <text class="title">欢迎使用</text>
    </view>
    
    <view class="form">
      <view class="input-group">
        <text class="label">手机号</text>
        <input
          v-model="form.phone"
          type="number"
          maxlength="11"
          placeholder="请输入手机号"
          class="input"
        />
      </view>
      
      <view class="input-group">
        <text class="label">密码</text>
        <input
          v-model="form.password"
          password
          placeholder="请输入密码"
          class="input"
        />
      </view>
      
      <button 
        class="login-btn" 
        :loading="loading"
        @click="handleLogin"
      >
        登录
      </button>
      
      <view class="actions">
        <text @click="goRegister">注册账号</text>
        <text @click="goForgot">忘记密码？</text>
      </view>
    </view>
  </view>
</template>

<script setup>
import { ref } from 'vue'
import { userApi } from '@/api/index.js'

const form = ref({
  phone: '',
  password: ''
})
const loading = ref(false)

const handleLogin = async () => {
  // 表单验证
  if (!form.value.phone) {
    uni.showToast({ title: '请输入手机号', icon: 'none' })
    return
  }
  if (!/^1[3-9]\d{9}$/.test(form.value.phone)) {
    uni.showToast({ title: '手机号格式错误', icon: 'none' })
    return
  }
  if (!form.value.password) {
    uni.showToast({ title: '请输入密码', icon: 'none' })
    return
  }
  
  loading.value = true
  
  try {
    const res = await userApi.login({
      phone: form.value.phone,
      password: form.value.password
    })
    
    // 保存登录状态
    uni.setStorageSync('token', res.data.token)
    uni.setStorageSync('userInfo', res.data.userInfo)
    
    uni.showToast({ title: '登录成功', icon: 'success' })
    
    setTimeout(() => {
      uni.switchTab({ url: '/pages/index/index' })
    }, 1500)
  } catch (err) {
    console.log('登录失败', err)
  } finally {
    loading.value = false
  }
}

const goRegister = () => {
  uni.navigateTo({ url: '/pages/register/register' })
}

const goForgot = () => {
  uni.navigateTo({ url: '/pages/forgot/forgot' })
}
</script>
```

## 五、后端接口约定（前后端配合）

### 统一返回格式

```json
// 成功
{
  "code": 200,
  "msg": "操作成功",
  "data": {
    "id": 1,
    "name": "张三"
  }
}

// 列表数据
{
  "code": 200,
  "msg": "success",
  "data": {
    "list": [...],
    "total": 100,
    "page": 1,
    "size": 10
  }
}

// 失败
{
  "code": 400,
  "msg": "参数错误：手机号不能为空"
}

// Token过期
{
  "code": 401,
  "msg": "登录已过期，请重新登录"
}
```

```
Content-Type: application/json
Authorization: Bearer eyJhbGciOiJIUzI1NiIs...
```
