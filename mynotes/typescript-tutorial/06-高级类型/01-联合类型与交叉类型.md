# 01 - 联合类型、交叉类型与类型守卫

---

## 一、联合类型（Union Types）

联合类型表示"可以是 A，也可以是 B"。

```typescript
// 用 | 连接多个类型
type StringOrNumber = string | number

let value: StringOrNumber = "hello"   // ✅
value = 42                             // ✅
// value = true                        // ❌ 只能是 string 或 number

// 实际应用场景

type Status = "pending" | "success" | "failed" | "cancelled"
let orderStatus: Status = "pending"     // ✅
// orderStatus = "done"                 // ❌ 不能是其他值


type Theme = "light" | "dark" | "auto"
type Size = "small" | "medium" | "large"

// 函数参数用联合类型
function printId(id: number | string) {
  console.log(`ID: ${id}`)
}

printId(101)           // ✅
printId("abc-123")     // ✅
```

### 联合类型的 narrowed（类型收窄）

当你有一个联合类型的变量时，TypeScript 不知道具体是哪个类型，所以只能用它们共有的属性和方法。

```typescript
function printValue(value: string | number) {
  // value.  // 只能提示 toString 等共有的方法

  // 需要先判断类型
  if (typeof value === "string") {
    // 在这个代码块里，TypeScript 知道 value 是 string
    console.log(value.toUpperCase())   // ✅ string 的方法
  } else {
    // 在这里，TypeScript 知道 value 是 number
    console.log(value.toFixed(2))      // ✅ number 的方法
  }
}
```

### 对象联合类型

```typescript
interface Square {
  kind: "square"
  size: number
}

interface Rectangle {
  kind: "rectangle"
  width: number
  height: number
}

interface Circle {
  kind: "circle"
  radius: number
}

// Shape 可以是三种之一
type Shape = Square | Rectangle | Circle

function getArea(shape: Shape): number {
  // 用 kind 字段区分具体类型（这叫"可辨识联合"）
  switch (shape.kind) {
    case "square":
      return shape.size * shape.size      // ✅ TS 知道 shape 是 Square
    case "rectangle":
      return shape.width * shape.height   // ✅ TS 知道 shape 是 Rectangle
    case "circle":
      return Math.PI * shape.radius ** 2  // ✅ TS 知道 shape 是 Circle
    default:
      // 如果有遗漏的 case，这里会报错
      const _exhaustiveCheck: never = shape
      return _exhaustiveCheck
  }
}
```

## 二、交叉类型（Intersection Types）

交叉类型表示"同时拥有 A 和 B 的所有属性"。

```typescript
// 用 & 连接多个类型
interface Person {
  name: string
  age: number
}

interface Employee {
  company: string
  salary: number
}

// EmployeePerson 同时拥有 Person 和 Employee 的所有属性
type EmployeePerson = Person & Employee

const emp: EmployeePerson = {
  name: "张三",
  age: 28,
  company: "科技公司",
  salary: 15000
}
```

### 属性冲突时

```typescript
interface A {
  foo: string
}

interface B {
  foo: number   // 和 A 的 foo 类型冲突
}

type C = A & B
// C 的 foo 类型变成 never（不可能同时是 string 和 number）
// 实际很少遇到，设计时避免
```

### 实际应用：扩展组件 Props

```typescript
// 基础按钮属性
interface BaseButtonProps {
  size?: "small" | "medium" | "large"
  disabled?: boolean
  onClick?: () => void
}

// 主按钮额外属性
interface PrimaryButtonProps {
  type: "primary"
  danger?: boolean
}

// 链接按钮额外属性
interface LinkButtonProps {
  type: "link"
  href: string
}

// 最终 Props 是基础和类型的交叉
type ButtonProps = BaseButtonProps & (PrimaryButtonProps | LinkButtonProps)

// 使用
const primaryBtn: ButtonProps = {
  type: "primary",
  size: "large",
  onClick: () => {}
}

const linkBtn: ButtonProps = {
  type: "link",
  href: "https://example.com",
  size: "small"
}
```

## 三、类型断言（Type Assertion）

当你比 TypeScript 更清楚一个值的类型时，可以用类型断言告诉 TS。

```typescript
// 场景：从 DOM 获取元素
const canvas = document.getElementById("canvas")
// canvas 的类型是 HTMLElement | null

// 你知道这个元素肯定是 canvas
canvas.getContext("2d")   // ❌ 报错：HTMLElement 可能没有 getContext

// 用 as 告诉 TS："相信我，这是 HTMLCanvasElement"
const canvas2 = document.getElementById("canvas") as HTMLCanvasElement
canvas2.getContext("2d")  // ✅

// 也可以用尖括号语法（JSX 中不能用这种）
const canvas3 = <HTMLCanvasElement>document.getElementById("canvas")
```

**注意**：类型断言不会真的转换类型，只是告诉 TS "别检查了，相信我"。如果断言错了，运行时还是会出错。

```typescript
const str: unknown = "hello"

// 安全的断言方式：先判断
if (typeof str === "string") {
  console.log(str.length)  // ✅
}

// 强制断言（不安全，尽量不用）
const num = str as number
// num 的类型是 number，但实际值是 "hello"
// 调 num.toFixed() 会运行时报错
```

## 四、类型守卫（Type Guards）

类型守卫是一些表达式，它们在运行时检查类型，同时让 TypeScript 在编译时缩小类型范围。

### 1. typeof 类型守卫

```typescript
function processValue(value: string | number | boolean) {
  if (typeof value === "string") {
    // value 被收窄为 string
    return value.toUpperCase()
  }

  if (typeof value === "number") {
    // value 被收窄为 number
    return value.toFixed(2)
  }

  // value 被收窄为 boolean
  return value ? "是" : "否"
}
```

### 2. instanceof 类型守卫

```typescript
class Dog {
  bark() { return "汪汪" }
}

class Cat {
  meow() { return "喵喵" }
}

function makeSound(animal: Dog | Cat) {
  if (animal instanceof Dog) {
    // animal 被收窄为 Dog
    return animal.bark()
  }

  // animal 被收窄为 Cat
  return animal.meow()
}
```

### 3. in 类型守卫

```typescript
interface Bird {
  fly: () => void
}

interface Fish {
  swim: () => void
}

function move(animal: Bird | Fish) {
  if ("fly" in animal) {
    // animal 被收窄为 Bird
    animal.fly()
  } else {
    // animal 被收窄为 Fish
    animal.swim()
  }
}
```

### 4. 自定义类型守卫函数

```typescript
// 定义一个类型守卫函数
function isString(value: unknown): value is string {
  return typeof value === "string"
}

function process(input: unknown) {
  if (isString(input)) {
    // 这里 input 被收窄为 string
    console.log(input.toUpperCase())
  }
}

// 更实用的例子：判断是否为有效的用户对象
interface User {
  id: number
  name: string
}

function isUser(obj: unknown): obj is User {
  return (
    typeof obj === "object" &&
    obj !== null &&
    "id" in obj &&
    "name" in obj &&
    typeof (obj as Record<string, unknown>).id === "number" &&
    typeof (obj as Record<string, unknown>).name === "string"
  )
}

const data = fetchDataFromServer()   // 返回 unknown

if (isUser(data)) {
  // data 被收窄为 User
  console.log(data.name)   // ✅ 安全
}
```

## 五、实用工具类型（Utility Types）

TypeScript 内置了很多好用的工具类型。

### Partial（全部变可选）

```typescript
interface User {
  id: number
  name: string
  email: string
  age: number
}

// Partial<User> 等价于：
// {
//   id?: number
//   name?: string
//   email?: string
//   age?: number
// }

function updateUser(id: number, data: Partial<User>) {
  // 可以只传部分字段
  updateToDatabase(id, data)
}

updateUser(1, { name: "新名字" })          // ✅
updateUser(1, { email: "new@example.com" }) // ✅
```

### Required（全部变必填）

```typescript
interface Config {
  host?: string
  port?: number
  ssl?: boolean
}

// Required<Config> 所有字段都变成必填
function validateConfig(config: Required<Config>) {
  // config.host, config.port, config.ssl 都一定有值
}
```

### Pick（挑选部分属性）

```typescript
interface User {
  id: number
  name: string
  email: string
  password: string
  age: number
}

// 只挑 name 和 email
type UserProfile = Pick<User, "name" | "email">

// 等价于：
// type UserProfile = {
//   name: string
//   email: string
// }

function showProfile(profile: UserProfile) {
  console.log(profile.name, profile.email)
}
```

### Omit（排除部分属性）

```typescript
// 排除 password
type SafeUser = Omit<User, "password">

// 等价于：
// type SafeUser = {
//   id: number
//   name: string
//   email: string
//   age: number
// }

// 返回给前端的数据不应该包含 password
function getUserForClient(): SafeUser {
  return { id: 1, name: "张三", email: "zs@example.com", age: 25 }
}
```

### Record（创建键值对类型）

```typescript
// Record<K, V> 表示"键类型为 K，值类型为 V"的对象

// 用户ID到用户名的映射
type UserMap = Record<number, string>

const users: UserMap = {
  1: "张三",
  2: "李四"
}

// 错误码到错误消息的映射
type ErrorMessages = Record<number, string>

const errors: ErrorMessages = {
  400: "请求参数错误",
  401: "未登录",
  403: "无权访问",
  500: "服务器错误"
}
```

### ReturnType（获取函数返回类型）

```typescript
function createUser() {
  return {
    id: 1,
    name: "张三",
    email: "zs@example.com"
  }
}

// 自动获取 createUser 的返回类型
type User = ReturnType<typeof createUser>

// 等价于：
// type User = { id: number; name: string; email: string }
```

### Parameters（获取函数参数类型）

```typescript
function updateUser(id: number, name: string, email: string) {
  // ...
}

// 获取参数类型元组
type UpdateUserParams = Parameters<typeof updateUser>

// 等价于：
// type UpdateUserParams = [number, string, string]

// 可以用作其他函数的参数类型
function logUpdate(...args: UpdateUserParams) {
  console.log("更新用户:", args)
}
```

## 六、工具类型速查表

| 工具类型 | 作用 | 示例 |
|---------|------|------|
| `Partial<T>` | 所有属性变可选 | `Partial<User>` |
| `Required<T>` | 所有属性变必填 | `Required<Config>` |
| `Readonly<T>` | 所有属性变只读 | `Readonly<User>` |
| `Pick<T, K>` | 挑选部分属性 | `Pick<User, "name">` |
| `Omit<T, K>` | 排除部分属性 | `Omit<User, "password">` |
| `Record<K, V>` | 键值对类型 | `Record<string, number>` |
| `Exclude<T, U>` | 从 T 中排除 U | `Exclude<"a"\|"b", "a">` → `"b"` |
| `Extract<T, U>` | 从 T 中提取 U | `Extract<"a"\|"b", "a"\|"c">` → `"a"` |
| `NonNullable<T>` | 排除 null 和 undefined | `NonNullable<string\|null>` → `string` |
| `ReturnType<T>` | 获取函数返回类型 | `ReturnType<typeof fn>` |
| `Parameters<T>` | 获取函数参数类型 | `Parameters<typeof fn>` |
