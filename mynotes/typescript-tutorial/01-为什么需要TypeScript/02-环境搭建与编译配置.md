# 02 - 环境搭建与 tsconfig.json 配置详解

---

## 一、安装 TypeScript

```bash
# 全局安装（可以在任意地方用 tsc 命令）
npm install -g typescript

# 验证安装
tsc --version   # 显示版本号，如 Version 5.4.5

# 项目内安装（推荐，每个项目用自己的版本）
npm install typescript --save-dev
```

---

## 二、第一个 TypeScript 程序

```bash
# 创建项目目录
mkdir ts-demo && cd ts-demo

# 初始化 npm 项目
npm init -y

# 安装 TypeScript
npm install typescript --save-dev
```

```typescript
// src/hello.ts
function greet(name: string): string {
  return `你好，${name}！`
}

const message: string = greet("张三")
console.log(message)
```

```bash
# 编译
tsc src/hello.ts

# 会生成 src/hello.js
# function greet(name) {
#   return "你好，" + name + "！"
# }
# var message = greet("张三")
# console.log(message)
```

```bash
# 运行编译后的 JS
node src/hello.js
# 输出：你好，张三！
```

## 三、tsconfig.json 配置详解

```bash
# 生成配置文件
tsc --init
```

```json
{
  "compilerOptions": {
    /* ────────────── 基础配置 ────────────── */
    "target": "ES2020",
    // 编译目标 JS 版本
    // ES5: 兼容旧浏览器
    // ES2020: 现代浏览器，支持 async/await、可选链等
    // ESNext: 最新特性

    "module": "ESNext",
    // 模块系统
    // CommonJS: Node.js 默认
    // ESNext: 现代模块化，配合 bundler 使用
    // UMD: 兼容浏览器和 Node

    "lib": ["ES2020", "DOM", "DOM.Iterable"],
    // 包含的类型定义库
    // ES2020: ES2020 的 API 类型
    // DOM: 浏览器 DOM API 类型（document, window 等）
    // DOM.Iterable: DOM 可迭代类型

    "jsx": "preserve",
    // JSX 处理方式（React/Vue 项目用）
    // preserve: 保留 JSX 语法
    // react: 编译为 React.createElement

    /* ────────────── 路径配置 ────────────── */
    "rootDir": "./src",
    // TS 源码根目录

    "outDir": "./dist",
    // 编译输出目录

    "baseUrl": ".",
    // 基础路径，用于下面的 paths

    "paths": {
      "@/*": ["src/*"]
    },
    // 路径别名，import { xxx } from '@/utils/xxx'
    // 需要配合 bundler（Vite/Webpack）使用

    /* ────────────── 类型检查严格程度 ────────────── */
    "strict": true,
    // 开启所有严格类型检查（强烈推荐开启）
    // 这是 TS 的灵魂，关闭了就失去意义

    "noImplicitAny": true,
    // 不允许隐式的 any 类型
    // 如果你忘了标注类型，TS 不会默认推断为 any，而是报错
    // ❌ function foo(x) { }  // 会报错，x 隐式推断为 any
    // ✅ function foo(x: number) { }

    "strictNullChecks": true,
    // 严格检查 null 和 undefined
    // 开启后，string 类型的变量不能赋值为 null
    // 需要用 string | null 才能存 null

    "strictFunctionTypes": true,
    // 严格检查函数参数类型

    /* ────────────── 额外检查 ────────────── */
    "noUnusedLocals": true,
    // 报错未使用的局部变量（帮助清理代码）

    "noUnusedParameters": true,
    // 报错未使用的函数参数

    "noImplicitReturns": true,
    // 函数所有分支必须有返回值

    "noFallthroughCasesInSwitch": true,
    // switch 的 case 必须有 break 或 return

    /* ────────────── 模块解析 ────────────── */
    "moduleResolution": "bundler",
    // 模块解析策略
    // node: Node.js 方式
    // bundler: 现代 bundler 方式（Vite 推荐）

    "esModuleInterop": true,
    // 兼容 CommonJS 和 ES Module 的混用
    // 允许 import foo from 'commonjs-module'

    "allowSyntheticDefaultImports": true,
    // 允许默认导入没有 default export 的模块

    "resolveJsonModule": true,
    // 允许 import json 文件

    "skipLibCheck": true,
    // 跳过所有 .d.ts 文件的类型检查
    // 加快编译速度，避免第三方库的类型冲突

    "forceConsistentCasingInFileNames": true,
    // 强制文件名大小写一致（避免跨平台问题）

    /* ────────────── 声明文件 ────────────── */
    "declaration": true,
    // 生成 .d.ts 声明文件（发布库时需要）

    "declarationDir": "./dist/types",
    // 声明文件输出目录

    /* ────────────── SourceMap ────────────── */
    "sourceMap": true
    // 生成 .map 文件，用于调试时映射回 TS 源码
  },

  /* ────────────── 包含的文件 ────────────── */
  "include": ["src/**/*"],
  // 编译哪些文件
  // src/**/*: src 目录下的所有 .ts 文件

  "exclude": ["node_modules", "dist", "**/*.test.ts"]
  // 排除哪些文件
  // 默认排除 node_modules 和 outDir
}
```

## 四、常用编译命令

```bash
# 编译整个项目（根据 tsconfig.json）
tsc

# 监视模式，文件变化自动编译
tsc --watch

# 只编译不输出（仅做类型检查）
tsc --noEmit

# 编译指定文件
tsc src/index.ts

# 指定配置文件
tsc --project tsconfig.build.json
```

## 五、Vue3 + TypeScript 项目的 tsconfig.json

```json
{
  "compilerOptions": {
    "target": "ES2020",
    "module": "ESNext",
    "lib": ["ES2020", "DOM", "DOM.Iterable"],
    "jsx": "preserve",
    "moduleResolution": "bundler",
    "resolveJsonModule": true,
    "types": ["vite/client"],
    "allowImportingTsExtensions": true,
    "strict": true,
    "noUnusedLocals": true,
    "noUnusedParameters": true,
    "noFallthroughCasesInSwitch": true,
    "esModuleInterop": true,
    "skipLibCheck": true,
    "forceConsistentCasingInFileNames": true
  },
  "include": ["src/**/*.ts", "src/**/*.tsx", "src/**/*.vue"]
}
```

## 六、快速检查清单

创建 TS 项目时的标准流程：

```bash
# 1. 初始化
npm init -y

# 2. 安装 TypeScript
npm install typescript --save-dev

# 3. 生成配置文件
tsc --init

# 4. 编辑 tsconfig.json
#    - target: ES2020
#    - strict: true（必须）
#    - rootDir: "./src"
#    - outDir: "./dist"
#    - paths: 配置别名

# 5. 安装类型定义（如果需要）
npm install @types/node --save-dev

# 6. 开始写代码
```
