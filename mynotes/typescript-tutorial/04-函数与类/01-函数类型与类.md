## 一、函数类型

### 1. 函数参数和返回值的类型标注

```typescript
// 基本函数
function add(a: number, b: number): number {
  return a + b
}

add(1, 2)        // ✅ 返回 3
add(1, "2")      // ❌ 第二个参数必须是 number
add(1, 2, 3)     // ❌ 参数太多

// 箭头函数
const multiply = (a: number, b: number): number => a * b

// 没有返回值的函数（void）
function logMessage(msg: string): void {
  console.log(msg)
  // 不需要 return
}

// 返回值可能是某个类型或 null
function findUser(id: number): User | null {
  const user = userList.find(u => u.id === id)
  return user || null
}
```

### 2. 可选参数和默认参数

```typescript
// 可选参数（?）
function greet(name: string, greeting?: string): string {
  // greeting 可能不存在，使用时要注意
  if (greeting) {
    return `${greeting}，${name}！`
  }
  return `你好，${name}！`
}

greet("张三")                    // ✅ "你好，张三！"
greet("张三", "早上好")           // ✅ "早上好，张三！"

// 默认参数
function greetWithDefault(name: string, greeting: string = "你好"): string {
  return `${greeting}，${name}！`
}

greetWithDefault("张三")          // ✅ "你好，张三！"

// 剩余参数
function sum(...numbers: number[]): number {
  return numbers.reduce((total, n) => total + n, 0)
}

sum(1, 2, 3, 4, 5)   // ✅ 返回 15
```

### 3. 函数类型作为变量类型

```typescript
// 定义一个"函数类型"：接收两个 number，返回 number
type MathOperation = (a: number, b: number) => number

// 用这个类型定义变量
const add: MathOperation = (a, b) => a + b
const subtract: MathOperation = (a, b) => a - b

// 回调函数的类型
function calculate(a: number, b: number, operation: MathOperation): number {
  return operation(a, b)
}

calculate(10, 5, add)       // ✅ 15
calculate(10, 5, subtract)  // ✅ 5
calculate(10, 5, (a, b) => a * b)  // ✅ 也可以传匿名函数
```

### 4. 函数重载

当同一个函数根据不同参数类型返回不同类型时，用函数重载。

```typescript
// 重载签名：声明函数的多种调用方式
function parseInput(input: string): number      // 传 string 返回 number
function parseInput(input: number): string      // 传 number 返回 string

// 实现签名
function parseInput(input: string | number): number | string {
  if (typeof input === "string") {
    return parseInt(input, 10)   // string → number
  }
  return input.toString()         // number → string
}

// 使用
const num = parseInput("123")     // num 的类型推断为 number
const str = parseInput(456)       // str 的类型推断为 string
```

**实际应用场景**：

```typescript
// 根据传入参数的不同，返回不同的数据结构
interface UserByIdResponse {
  id: number
  name: string
}

interface UserListResponse {
  list: UserByIdResponse[]
  total: number
}

// 重载
function fetchUser(id: number): Promise<UserByIdResponse>
function fetchUser(params: { page: number; size: number }): Promise<UserListResponse>

// 实现
function fetchUser(input: number | { page: number; size: number }) {
  if (typeof input === "number") {
    return api.get(`/users/${input}`)
  }
  return api.get("/users", { params: input })
}

// 使用
const user = await fetchUser(1)           // user 的类型是 UserByIdResponse
const list = await fetchUser({ page: 1, size: 10 })  // list 的类型是 UserListResponse
```

## 二、类（Class）

### 1. 基本类定义

```typescript
class User {
  // 属性声明
  id: number
  name: string
  email: string

  // 构造函数
  constructor(id: number, name: string, email: string) {
    this.id = id
    this.name = name
    this.email = email
  }

  // 方法
  greet(): string {
    return `你好，我是${this.name}`
  }

  updateEmail(newEmail: string): void {
    this.email = newEmail
  }
}

// 使用
const user = new User(1, "张三", "zs@example.com")
console.log(user.greet())     // "你好，我是张三"
user.updateEmail("new@example.com")
```

### 2. 简化写法（构造函数参数直接声明属性）

```typescript
class User {
  // 在构造函数参数前加修饰符，会自动创建同名属性
  constructor(
    public id: number,
    public name: string,
    private email: string    // private：只能在类内部访问
  ) {}

  getEmail(): string {
    // 类内部可以访问 private 属性
    return this.email
  }
}

const user = new User(1, "张三", "zs@example.com")
console.log(user.name)        // ✅ "张三"
console.log(user.id)          // ✅ 1
console.log(user.email)       // ❌ 报错：email 是私有的
console.log(user.getEmail())  // ✅ 通过公共方法获取
```

### 3. 访问修饰符

```typescript
class Employee {
  public name: string         // public：任何地方都能访问（默认）
  private salary: number      // private：只能在类内部访问
  protected department: string // protected：类内部 + 子类可以访问
  readonly id: number         // readonly：创建后不能修改

  constructor(name: string, salary: number, department: string, id: number) {
    this.name = name
    this.salary = salary
    this.department = department
    this.id = id
  }
}

class Manager extends Employee {
  constructor(name: string, salary: number, department: string, id: number) {
    super(name, salary, department, id)
  }

  getInfo(): string {
    // 可以访问 public 和 protected
    return `${this.name} - ${this.department}`
    // 不能访问 private：this.salary ❌
  }
}

const emp = new Employee("张三", 10000, "技术部", 1)
console.log(emp.name)         // ✅ public
console.log(emp.salary)       // ❌ private
console.log(emp.department)   // ❌ protected
```

| 修饰符 | 类内部 | 子类 | 外部 |
|--------|--------|------|------|
| public | ✅ | ✅ | ✅ |
| protected | ✅ | ✅ | ❌ |
| private | ✅ | ❌ | ❌ |
| readonly | 只能读 | - | - |

### 4. 类实现接口

```typescript
// 定义接口（规定"应该有什么能力"）
interface CanFly {
  fly(): void
}

interface CanSwim {
  swim(): void
}

// 类实现接口（"我承诺有这些能力"）
class Duck implements CanFly, CanSwim {
  fly() {
    console.log("鸭子在飞")
  }

  swim() {
    console.log("鸭子在游泳")
  }
}

// 如果漏实现了接口的方法，会报错
class Chicken implements CanFly {
  // ❌ 报错：类没有实现接口的 fly 方法
}
```

**实际应用**：Repository 模式

```typescript
interface IUserRepository {
  findById(id: number): Promise<User | null>
  findAll(): Promise<User[]>
  save(user: User): Promise<void>
  delete(id: number): Promise<void>
}

class UserRepository implements IUserRepository {
  async findById(id: number): Promise<User | null> {
    // 从数据库查询
    return db.users.findOne({ id })
  }

  async findAll(): Promise<User[]> {
    return db.users.findAll()
  }

  async save(user: User): Promise<void> {
    await db.users.upsert(user)
  }

  async delete(id: number): Promise<void> {
    await db.users.delete({ id })
  }
}
```

### 5. 抽象类

```typescript
// 抽象类：不能直接 new，只能被继承
abstract class Animal {
  constructor(public name: string) {}

  // 抽象方法：子类必须实现
  abstract makeSound(): void

  // 普通方法：子类可以直接用
  move(): void {
    console.log(`${this.name}在移动`)
  }
}

// 抽象类不能被实例化
// const animal = new Animal("动物")   // ❌ 报错

class Dog extends Animal {
  makeSound() {
    console.log("汪汪")
  }
}

class Cat extends Animal {
  makeSound() {
    console.log("喵喵")
  }
}

const dog = new Dog("旺财")
dog.makeSound()   // "汪汪"
dog.move()        // "旺财在移动"
```

### 6. 泛型类

```typescript
// 泛型类：类可以处理多种类型
class Container<T> {
  private items: T[] = []

  add(item: T): void {
    this.items.push(item)
  }

  get(index: number): T | undefined {
    return this.items[index]
  }

  getAll(): T[] {
    return [...this.items]
  }
}

// 存储数字
const numberContainer = new Container<number>()
numberContainer.add(1)
numberContainer.add(2)
// numberContainer.add("3")   // ❌ 报错

// 存储用户
const userContainer = new Container<User>()
userContainer.add({ id: 1, name: "张三", email: "" })
// userContainer.add(123)     // ❌ 报错
```
