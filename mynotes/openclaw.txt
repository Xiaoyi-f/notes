curl -SO https://openclaw.ai/install.sh | bash  # 需要node版本在22及以上
openclaw --version
openclaw doctor 健康检查
# 提示command not found则配置系统环境变量
export PATH="$(npm prefix -g)/bin:$PATH"
# 获取对应模型的 API Key
openclaw onboard 进行接入模型
openclaw tui 进入对话
# 启动服务
openclaw gateway start

openclaw gateway status
openclaw gateway logs

# 服务器安全组规则管控，一般可能需要放行18789端口等
在 openclaw tui 进入对话后
/status 指令可以查看模型、Token消耗
/new 清空上下文，新开对话
/model 模型 切换模型
openclaw sessions list 列出所有会话进行管理

openclaw skills install 技能
openclaw skills list

# qq开放平台中可以实现qq机器人接入openclaw
# 微信官方开放 @tencent-weixin/openclaw-weixin-cli@latest node插件 可以实现微信接入openclaw
